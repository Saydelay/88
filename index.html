#!/usr/bin/env bash # ===================================================================== # SAYDELAY – Videos fürs Streaming optimieren # Einmal ausführen: sudo bash optimieren.sh # Danach prüft der Pi automatisch alle 10 Minuten neue Filme. # # Was passiert: Das "Inhaltsverzeichnis" (moov) wird an den Anfang der # Datei gesetzt -> Film startet sofort, Vorspulen geht direkt. # MKV-Dateien mit H.264 werden zu MP4. Bildqualität bleibt gleich. # ===================================================================== set -e if [ "$EUID" -ne 0 ]; then echo "Bitte mit sudo starten: sudo bash optimieren.sh"; exit 1; fi SELF=/usr/local/bin/saydelay-optimieren MEDIA=$(readlink -f /srv/saydelay/media) OWNER=$(stat -c %U "$MEDIA") # ---------- Einmalige Installation ---------- if [ "$(readlink -f "$0")" != "$SELF" ]; then echo "==> ffmpeg installieren (dauert auf dem Pi 3 ein paar Minuten)" apt-get install -y ffmpeg python3 install -m 755 "$0" "$SELF" cat > /etc/systemd/system/saydelay-optimieren.service < /etc/systemd/system/saydelay-optimieren.timer </dev/null || mkdir -p "$MEDIA/Covers" echo "==> Ordner 'Covers' angelegt." case "$0" in /srv/saydelay/www/*) rm -f "$0" ;; esac echo "==> Automatik ist an. Jetzt erster Durchlauf über alle Filme:" echo "" fi # ---------- Nur einmal gleichzeitig laufen ---------- exec 9>/run/saydelay-optimieren.lock flock -n 9 || { echo "Optimierung läuft gerade schon."; exit 0; } # Steht das Inhaltsverzeichnis (moov) schon vor den Videodaten (mdat)? faststart_ok() { python3 - "$1" <<'PY' import sys, os, struct p = sys.argv[1]; size = os.path.getsize(p); pos = 0 with open(p, 'rb') as f: while pos + 8 <= size: f.seek(pos); n, t = struct.unpack('>I4s', f.read(8)) if n == 1: n = struct.unpack('>Q', f.read(8))[0] elif n == 0: n = size - pos if t == b'moov': sys.exit(0) if t == b'mdat': sys.exit(1) if n < 8: break pos += n sys.exit(1) PY } count=0 find "$MEDIA" -type f \( -iname '*.mp4' -o -iname '*.m4v' -o -iname '*.mkv' -o -iname '*.mov' -o -iname '*.avi' \) \ -not -name '*.saydelay-tmp.mp4' -mmin +2 -print0 | while IFS= read -r -d '' f; do ext=$(echo "${f##*.}" | tr 'A-Z' 'a-z') if [ "$ext" = "mp4" ] || [ "$ext" = "m4v" ]; then faststart_ok "$f" && continue fi v=$(ffprobe -v error -select_streams v:0 -show_entries stream=codec_name -of csv=p=0 "$f" /dev/null || true mv -f "$tmp" "$out" [ "$f" != "$out" ] && rm -f "$f" echo " ✓ fertig" count=$((count+1)) else rm -f "$tmp" echo " ✗ Fehler – Datei bleibt unverändert" fi done echo "" echo "Optimierung abgeschlossen."
