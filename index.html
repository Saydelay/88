#!/usr/bin/env bash # ===================================================================== # SAYDELAY – Videos fürs Streaming optimieren # Einmal ausführen: sudo bash optimieren.sh # Danach prüft der Pi automatisch alle 10 Minuten neue Filme. # # Was passiert: Das "Inhaltsverzeichnis" (moov) wird an den Anfang der # Datei gesetzt -> Film startet sofort, Vorspulen geht direkt. # MKV-Dateien mit H.264 werden zu MP4. Bildqualität bleibt gleich. # ===================================================================== set -e if [ "$EUID" -ne 0 ]; then echo "Bitte mit sudo starten: sudo bash optimieren.sh"; exit 1; fi SELF=/usr/local/bin/saydelay-optimieren MEDIA=$(readlink -f /srv/saydelay/media) OWNER=$(stat -c %U "$MEDIA") # ---------- Einmalige Installation ---------- if [ "$(readlink -f "$0")" != "$SELF" ]; then echo "==> ffmpeg installieren (dauert auf dem Pi 3 ein paar Minuten)" apt-get install -y ffmpeg python3 inotify-tools install -m 755 "$0" "$SELF" cat > /etc/systemd/system/saydelay-optimieren.service < /etc/systemd/system/saydelay-optimieren.timer < /usr/local/bin/saydelay-watch <<'WATCH' #!/usr/bin/env bash MEDIA=$(readlink -f /srv/saydelay/media) IGNORE='/\.thumbs/|\.saydelay-tmp\.|\.part$|\.tmp\.jpg$|/\.library\.json' EVENTS=close_write,moved_to,delete,create while true; do inotifywait -qq -r -e $EVENTS --exclude "$IGNORE" "$MEDIA" || { sleep 30; continue; } # warten, bis 5 Sekunden lang nichts mehr passiert (Kopieren fertig) while inotifywait -qq -r -t 5 -e $EVENTS --exclude "$IGNORE" "$MEDIA"; do :; done /usr/local/bin/saydelay-library 2>/dev/null # Filmliste sofort aktuell systemctl start --no-block saydelay-optimieren.service done WATCH chmod 755 /usr/local/bin/saydelay-watch cat > /etc/systemd/system/saydelay-watch.service < /etc/sysctl.d/90-saydelay.conf sysctl -q --system || true systemctl daemon-reload systemctl enable --now saydelay-optimieren.timer systemctl enable saydelay-watch systemctl restart saydelay-watch echo "==> Sofort-Erkennung ist an." sudo -u "$OWNER" mkdir -p "$MEDIA/Covers" 2>/dev/null || mkdir -p "$MEDIA/Covers" # Inhaltsverzeichnis: eine Datei mit allen Ordnern -> Seite braucht nur 1 Anfrage statt vieler cat > /usr/local/bin/saydelay-library <<'LIB' #!/usr/bin/env python3 import email.utils, json, os MEDIA = os.path.realpath('/srv/saydelay/media') OUT = os.path.join(MEDIA, '.library.json') def entries(d): out = [] try: names = sorted(os.listdir(d)) except Exception: return out for n in names: if n.startswith('.') or n.endswith('.part') or '.saydelay-tmp.' in n: continue p = os.path.join(d, n) try: st = os.stat(p) except Exception: continue isdir = os.path.isdir(p) e = {"name": n, "type": "directory" if isdir else "file", "mtime": email.utils.formatdate(st.st_mtime, usegmt=True)} if not isdir: e["size"] = st.st_size out.append(e) return out lib = {"": entries(MEDIA)} for top in ('Filme', 'Serien', 'Covers'): base = os.path.join(MEDIA, top) if not os.path.isdir(base): continue for root, dirs, files in os.walk(base): dirs[:] = sorted(d for d in dirs if not d.startswith('.')) rel = os.path.relpath(root, MEDIA).replace(os.sep, '/') lib[rel] = entries(root) if rel.count('/') >= 2: dirs[:] = [] tmp = OUT + '.tmp' with open(tmp, 'w') as f: json.dump({"v": 1, "dirs": lib}, f, ensure_ascii=False) try: os.chmod(tmp, 0o644) except Exception: pass os.replace(tmp, OUT) LIB chmod 755 /usr/local/bin/saydelay-library # Dateien haben jetzt Versionsnummern in der Adresse -> Browser dürfen sie dauerhaft speichern NGCONF=/etc/nginx/sites-available/saydelay if [ -f "$NGCONF" ]; then sed -i -e 's|add_header Cache-Control "public, max-age=86400";|add_header Cache-Control "public, max-age=31536000, immutable";|' \ -e 's|add_header Cache-Control "no-cache" always;|add_header Cache-Control "public, max-age=31536000, immutable";|' "$NGCONF" nginx -t && systemctl reload nginx && echo "==> Webserver: Bilder werden jetzt im Browser gespeichert (schneller)." fi echo "==> Ordner 'Covers' angelegt." case "$0" in /srv/saydelay/www/*) rm -f "$0" ;; esac echo "==> Automatik ist an. Jetzt erster Durchlauf über alle Filme:" echo "" fi # ---------- Nur einmal gleichzeitig laufen ---------- exec 9>/run/saydelay-optimieren.lock flock -n 9 || { echo "Optimierung läuft gerade schon."; exit 0; } # Steht das Inhaltsverzeichnis (moov) schon vor den Videodaten (mdat)? faststart_ok() { python3 - "$1" <<'PY' import sys, os, struct p = sys.argv[1]; size = os.path.getsize(p); pos = 0 with open(p, 'rb') as f: while pos + 8 <= size: f.seek(pos); n, t = struct.unpack('>I4s', f.read(8)) if n == 1: n = struct.unpack('>Q', f.read(8))[0] elif n == 0: n = size - pos if t == b'moov': sys.exit(0) if t == b'mdat': sys.exit(1) if n < 8: break pos += n sys.exit(1) PY } # ---------- Vorschaubilder (Covers sauber auf 600 px verkleinern) ---------- THUMBS="$MEDIA/.thumbs" mkdir -p "$THUMBS" find "$MEDIA" -path "$THUMBS" -prune -o -type f \( -iname '*.jpg' -o -iname '*.jpeg' -o -iname '*.png' -o -iname '*.webp' \) -print0 | while IFS= read -r -d '' img; do rel="${img#$MEDIA/}" sig="$(stat -c '%s-%Y' "$img")" # Größe-Datum = Versionsnummer th="$THUMBS/$rel.$sig.jpg" # neue Version = neuer Dateiname [ -f "$th" ] && continue mkdir -p "$(dirname "$th")" if ffmpeg -nostdin -v error -y -i "$img" -frames:v 1 \ -vf "scale='min(600,iw)':-2:flags=lanczos,format=yuvj420p" -q:v 3 "$th.tmp.jpg"; then mv -f "$th.tmp.jpg" "$th" echo "Vorschaubild: $rel" else rm -f "$th.tmp.jpg" fi done # Alte Versionen und Vorschaubilder von gelöschten Covers aufräumen find "$THUMBS" -type f -print0 | while IFS= read -r -d '' th; do rel="${th#$THUMBS/}" if [[ "$rel" =~ ^(.*)\.([0-9]+-[0-9]+)\.jpg$ ]]; then src="$MEDIA/${BASH_REMATCH[1]}" [ -f "$src" ] && [ "$(stat -c '%s-%Y' "$src")" = "${BASH_REMATCH[2]}" ] && continue fi rm -f "$th" done find "$THUMBS" -mindepth 1 -type d -empty -delete 2>/dev/null || true chmod -R a+rX "$THUMBS" 2>/dev/null || true /usr/local/bin/saydelay-library 2>/dev/null || true # Welche Spuren hat die Datei? -> "index codec tag pixfmt alle_ton_aac" probe() { ffprobe -v error -show_entries stream=index,codec_type,codec_name,codec_tag_string,pix_fmt -of json "$1" > "$DONE"; } find "$MEDIA" -path "$THUMBS" -prune -o -type f \( -iname '*.mp4' -o -iname '*.m4v' -o -iname '*.mkv' -o -iname '*.mov' -o -iname '*.avi' \) \ -not -name '*.saydelay-tmp.mp4' -mmin +2 -print0 | while IFS= read -r -d '' f; do grep -qxF "$f|$(stat -c '%s-%Y' "$f")" "$DONE" && continue name="${f#$MEDIA/}" read -r vidx v vtag pix allaac <<< "$(probe "$f" || echo '- error - - 1')" case "$v" in h264) if [[ "$pix" == *10* || "$pix" == *12* ]]; then echo "⚠ $name: 10-Bit-H.264 kann Safari nicht abspielen – bitte am PC mit umwandeln.bat umwandeln." mark_done "$f"; continue fi VTAG="" ;; hevc) VTAG="-tag:v hvc1" ;; # Apple braucht die Kennung "hvc1", sonst lädt Safari ewig *) echo "⚠ $name: Videoformat '$v' geht nicht im Browser – bitte am PC mit umwandeln.bat umwandeln." mark_done "$f"; continue ;; esac ext=$(echo "${f##*.}" | tr 'A-Z' 'a-z') need="" [ "$ext" != "mp4" ] && [ "$ext" != "m4v" ] && need="Format $ext" [ -z "$need" ] && ! faststart_ok "$f" && need="Inhaltsverzeichnis am Ende" [ -z "$need" ] && [ "$v" = "hevc" ] && [ "$vtag" != "hvc1" ] && need="HEVC-Kennung für Apple" [ -z "$need" ] && [ "$allaac" != "1" ] && need="Tonformat" if [ -z "$need" ]; then mark_done "$f"; continue; fi if [ "$allaac" = "1" ]; then AUDIO="-c:a copy"; else AUDIO="-c:a aac -b:a 160k -ac 2"; fi out="${f%.*}.mp4" tmp="${f%.*}.saydelay-tmp.mp4" echo "Optimiere: $name ($need)" if ffmpeg -nostdin -v error -y -i "$f" -map "0:$vidx" -map '0:a?' -c:v copy $VTAG $AUDIO -movflags +faststart "$tmp"; then touch -r "$f" "$tmp" chown --reference="$f" "$tmp" 2>/dev/null || true mv -f "$tmp" "$out" [ "$f" != "$out" ] && rm -f "$f" mark_done "$out" echo " ✓ fertig" else rm -f "$tmp" echo " ✗ Fehler – Datei bleibt unverändert" fi done /usr/local/bin/saydelay-library 2>/dev/null || true echo "" echo "Optimierung abgeschlossen."
